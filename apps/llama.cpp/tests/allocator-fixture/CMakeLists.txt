cmake_minimum_required(VERSION 3.19)
project("llama.cpp" C)
add_subdirectory(ggml-cpu)

# The wrapper's deferred injection runs before this verification callback.
function(check_allocator_graph)
    foreach(target ggml-cpu ggml-cpu-variant)
        get_target_property(sources "${target}" SOURCES)
        if(NOT sources STREQUAL XSAI_ALLOC_SRC)
            message(FATAL_ERROR "${target} did not select MNN: ${sources}")
        endif()
        get_target_property(includes "${target}" INCLUDE_DIRECTORIES)
        get_filename_component(expected "${XSAI_ALLOC_HEADER}" DIRECTORY)
        list(GET includes 0 actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "${target} did not prefer MNN headers")
        endif()
    endforeach()
    get_target_property(unrelated unrelated SOURCES)
    if(unrelated STREQUAL XSAI_ALLOC_SRC)
        message(FATAL_ERROR "Unrelated target was modified")
    endif()
endfunction()
cmake_language(DEFER CALL check_allocator_graph)
